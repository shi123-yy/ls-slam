        0x61, 0x74, 0x65, 0x20, 0x31, 0x5D, 0xE0, 0x0E"

        "0xA0, 0x0A, 0x12, 0x73, 0x4D, 0x4E, 0x20, 0x6D, 0x43, 0x68, 0x61, 0x6E, 0x67, 0x65, 0x53, 0x74,
        0x61, 0x74, 0x65, 0x20, 0x32, 0x5E, 0xE0, 0x0E"
3. 定位指令：
      "0xA0, 0x0A, 0x12, 0x73, 0x4D, 0x4E, 0x20, 0x6D, 0x43, 0x68, 0x61, 0x6E, 0x67, 0x65, 0x53, 0x74,

      "0xA0, 0x0A, 0x12, 0x73, 0x4D, 0x4E, 0x20, 0x6D, 0x43, 0x68, 0x61, 0x6E, 0x67, 0x65, 0x53, 0x74,
